    /* ═══════════ SAVE / UPDATE ═══════════ */
    function save() {
      savings = Math.min(Math.max(savings, 0), GOAL);
      localStorage.setItem("omerSavings", savings);
      localStorage.setItem("omerHistory", JSON.stringify(history));
      updateStreak();
      updateUI();
    }

    function addHistory(amount, desc) {
      history.unshift({ amount, desc, time: Date.now() });
      if (history.length > 200) history.pop();
    }

    function updateStreak() {
      const today = new Date().toDateString();
      if (streak.lastDate === today) return;
      const yesterday = new Date(Date.now() - 86400000).toDateString();
      streak.count = (streak.lastDate === yesterday) ? streak.count + 1 : 1;
      streak.lastDate = today;
      localStorage.setItem("omerStreak", JSON.stringify(streak));
      document.getElementById("streakCount").textContent = streak.count;
    }

    /* ═══════════ UI UPDATE ═══════════ */
    function updateUI() {
      const pct       = Math.floor((savings / GOAL) * 100);
      const remaining = Math.max(GOAL - savings, 0);

      // text stats
      document.getElementById("saved").textContent     = fmt(savings) + " ₪";
      document.getElementById("remaining").textContent = fmt(remaining) + " ₪";
      document.getElementById("progressPct").textContent = pct + "%";

      // daily average
      const days = Math.max(history.filter(h => h.amount > 0).length, 1);
      const totalEarned = history.filter(h => h.amount > 0).reduce((s,h) => s + h.amount, 0);
      const avgPerAction = Math.round(totalEarned / days);
      document.getElementById("dailyAvg").textContent = avgPerAction + " ₪";

      // ring
      const offset = CIRCUMFERENCE - (pct / 100) * CIRCUMFERENCE;
      document.getElementById("ringCircle").style.strokeDashoffset = offset;
      document.getElementById("ringPct").textContent = pct + "%";

      // bar
      document.getElementById("progressBar").style.width = pct + "%";

      // celebration
      document.getElementById("celebration").style.display = savings >= GOAL ? "block" : "none";
      if (savings >= GOAL && !confettiActive) launchConfetti();

      // next milestone hint
      const nextMs = [50,100,150,200,250,300,350,400,450,500,550,600,650,700,750,800,850,900,950,1000,1050,1100,1150,1200]
        .find(m => m > savings);
      const hint = document.getElementById("nextHint");
      if (nextMs) {
        const left = nextMs - savings;
        hint.innerHTML = עוד <span>${fmt(left)} ₪</span> לאבן הדרך הבאה: <span>${fmt(nextMs)} ₪</span>;
      } else {
        hint.innerHTML = 🏆 כל אבני הדרך הושלמו!;
      }

      // streak
      document.getElementById("streakCount").textContent = streak.count;

      renderMilestones();
      renderBadges();
      renderHistory();
    }

    function fmt(n) { return Number(n).toLocaleString('he-IL'); }

    /* ═══════════ MILESTONES ═══════════ */
    function renderMilestones() {
      const box = document.getElementById("milestones");
      box.innerHTML = "";
      for (let i = 1; i <= 24; i++) {
        const amt  = i * 50;
        const done = savings >= amt;
        const next = !done && savings >= (i - 1) * 50;
        const div  = document.createElement("div");
        div.className = "milestone" + (done ? " done" : next ? " next" : "");
        div.innerHTML  = done ? ✅<br>${amt}₪ : next ? 🎯<br>${amt}₪ : 🔒<br>${amt}₪;
        box.appendChild(div);
      }
    }

    /* ═══════════ BADGES ═══════════ */
    function renderBadges() {
      const box = document.getElementById("badges");
      box.innerHTML = "";
      ACHIEVEMENTS.forEach(([icon, title, amount]) => {
        const unlocked = savings >= amount;
        const badge = document.createElement("div");
        badge.className = "badge" + (unlocked ? " unlocked" : "");
        badge.innerHTML = <span class="badge-icon">${unlocked ? icon : "🔒"}</span>${title};
        box.appendChild(badge);
      });
    }

    /* ═══════════ MISSIONS ═══════════ */

      // filter buttons
      const filterBox = document.getElementById("missionFilter");
      filterBox.innerHTML = "";
      const cats = [["all","הכל"],["daily","יומי"],["weekly","שבועי"],["special","מיוחד"]];
      cats.forEach(([val, label]) => {
        const btn = document.createElement("button");
        btn.className = "filter-btn" + (activeCat === val ? " active" : "");
        btn.textContent = label;
        btn.onclick = () => { activeCat = val; renderMissions(); };
        filterBox.appendChild(btn);
      });

      // mission cards
      const box = document.getElementById("missions");
      box.innerHTML = "";
      const filtered = MISSIONS.filter(([,,cat]) => activeCat === "all" || cat === activeCat);
      filtered.forEach(([name, reward, cat]) => {
        const catLabels = { daily:"יומי", weekly:"שבועי", special:"מיוחד" };
        const card = document.createElement("div");
        card.className = "mission";

        box.appendChild(card);
      });

    /* ═══════════ HISTORY ═══════════ */

      history.forEach(h => {
        const d = new Date(h.time);
        const time = d.toLocaleDateString('he-IL') + " " + d.toLocaleTimeString('he-IL', {hour:'2-digit',minute:'2-digit'});

        item.innerHTML = `

      });

    /* ═══════════ ACTIONS ═══════════ */
    function completeMission(btn, name, amount) {
      const card = btn.closest(".mission");
      card.classList.add("completing");
      setTimeout(() => card.classList.remove("completing"), 500);

      addHistory(amount, ✅ ${name});

    function setManualAmount() {

    /* ═══════════ CONFETTI ═══════════ */
    function launchConfetti() {
      confettiActive = true;
      const canvas = document.getElementById("confettiCanvas");
      const ctx = canvas.getContext("2d");
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;

      const colors = ["#28ff8a","#00d9ff","#9b5cff","#ffe94d","#ff9f1c","#ff4d6d"];
      const pieces = Array.from({length: 160}, () => ({
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height - canvas.height,
        size: Math.random() * 10 + 4,
        color: colors[Math.floor(Math.random() * colors.length)],
        speed: Math.random() * 3 + 2,
        angle: Math.random() * 360,
        spin: Math.random() * 6 - 3,
        drift: Math.random() * 2 - 1,
      }));

      let frame = 0;
      function draw() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        pieces.forEach(p => {
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate((p.angle * Math.PI) / 180);
          ctx.fillStyle = p.color;
          ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
          ctx.restore();
          p.y += p.speed;
          p.x += p.drift;
          p.angle += p.spin;
        });
        frame++;
        if (frame < 220) requestAnimationFrame(draw);
        else { ctx.clearRect(0, 0, canvas.width, canvas.height); confettiActive = false; }
      }
      draw();
    }
